import { chromium } from "/home/user/read-it-well/node_modules/playwright-core/index.mjs";
import { writeFileSync, readFileSync } from "node:fs";
const OUT = process.argv[2];
const F = [
  ["s1-wallet-transactions", "/preview/session-b/sweep-wallet/transactions", ".nf-chip[aria-pressed=true],.nf-chip[aria-checked=true]", 1],
  ["s1-listing-day-slots", "/preview/session-b/sweep-home/listing-parts", ".nf-chip[aria-pressed=true],.nf-chip[aria-checked=true]", 1],
  ["s2-payments-toggle", "/preview/session-b/admin-money/payments", ".nf-md-toggle[aria-pressed=true]", 1],
  ["s3-escrow-file-this", "/preview/session-b/sweep-stays/escrow-detail", ".nf-btn--primary:disabled", 0],

  ["s3-assistant-send", "/preview/session-b/sweep-home/assistant", ".nf-btn--primary:disabled", 0],
  ["s3-thread-send", "/preview/session-b/sweep-social?f=plain", ".nf-composer__send:disabled", 0],

  ["s6-inbox-count", "/preview/session-b/sweep-social?f=inbox", ".nf-segmented,[role=tablist]", 0],

];
const b = await chromium.launch({ executablePath: "/opt/pw-browsers/chromium" });
async function crop(base, route, sel, parent, w, file) {
  const ctx = await b.newContext({ viewport: { width: w, height: w === 390 ? 844 : 900 }, deviceScaleFactor: 2, colorScheme: "dark", reducedMotion: "reduce" });
  const p = await ctx.newPage();
  await p.goto(base + route, { waitUntil: "load" }); await p.waitForTimeout(1200);
  let el = await p.$(sel);
  for (let i = 0; el && i < parent; i++) el = await el.evaluateHandle((e) => e.parentElement);
  let out = null;
  if (el && (await el.boundingBox())) {
    await el.evaluate((e) => e.scrollIntoView({ block: "center" }));
    await p.waitForTimeout(200);
    const bb = await el.boundingBox();
    const pad = 20;
    const x = Math.max(0, bb.x - pad), y = Math.max(0, bb.y - pad);
    const width = Math.min(w - x, bb.width + 2 * pad), height = Math.min(600, bb.height + 2 * pad + 12);
    await p.screenshot({ path: file, clip: { x, y, width, height } });
    out = file;
  }
  await ctx.close();
  return out;
}
for (const w of [390, 1440]) {
  for (const [name, route, sel, parent] of F) {
    const bf = await crop("http://127.0.0.1:3391", route, sel, parent, w, `${OUT}/${name}-${w}-before.png`);
    const af = await crop("http://127.0.0.1:3392", route, sel, parent, w, `${OUT}/${name}-${w}-after.png`);
    if (!bf || !af) { console.log("missing", name, w, !!bf, !!af); continue; }
    const img = (f) => "data:image/png;base64," + readFileSync(f).toString("base64");
    const html = `<html><body style="margin:0;background:#000814;color:#cfe3ff;font:14px system-ui;display:flex;gap:16px;padding:12px;align-items:flex-start">
      <div><div style="margin-bottom:6px">Before (main e93bb8a9)</div><img style="max-width:${w === 390 ? 390 : 700}px" src="${img(bf)}"></div>
      <div><div style="margin-bottom:6px">After</div><img style="max-width:${w === 390 ? 390 : 700}px" src="${img(af)}"></div></body></html>`;
    const ctx = await b.newContext({ viewport: { width: w === 390 ? 830 : 1450, height: 400 }, deviceScaleFactor: 1 });
    const p = await ctx.newPage();
    await p.setContent(html); await p.waitForTimeout(100);
    await p.screenshot({ path: `${OUT}/${name}-${w}.jpg`, fullPage: true, type: "jpeg", quality: 80 });
    await ctx.close();
    console.log("ok", name, w);
  }
}
await b.close();
