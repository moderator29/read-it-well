// S9 remainder: the gated host routes and /styleguide, signed in as the QA member (reads only),
// plus the host wizard's later steps on the audit2-fixes harness. Credentials from env only.
// usage: node final-s9.mjs <port> <outDir> <out.json>
import { chromium } from "/tmp/claude-0/-home-user-read-it-well/b3cd0ca7-526a-577f-99c1-fc3870877a0e/scratchpad/wt-final/node_modules/playwright-core/index.mjs";
import { writeFileSync, mkdirSync } from "node:fs";
const [port, dir, out] = process.argv.slice(2);
const BASE = `http://127.0.0.1:${port}`;
mkdirSync(dir, { recursive: true });
const b = await chromium.launch({ executablePath: "/opt/pw-browsers/chromium" });
const res = { at: new Date().toISOString(), live: {}, harness: {} };
const MEASURE = `(() => {
  const legacy = document.querySelectorAll(".nf-glass--card, .nf-card:not(.nf-panel)").length;
  const panels = document.querySelectorAll(".nf-panel").length;
  const prim = [...document.querySelectorAll(".nf-btn--primary")];
  const unlit = prim.filter((e) => !getComputedStyle(e).backgroundImage.includes("gradient")).length;
  const h1 = (document.querySelector("h1")?.textContent || "").trim().slice(0, 60);
  const overflow = document.documentElement.scrollWidth - innerWidth;
  const caps = [...document.querySelectorAll("button, a.nf-btn, [role=tab], input, select, .nf-badge")].filter((e) => { const r = e.getBoundingClientRect(); if (r.width < 1 || !(e.innerText || e.value || "").trim()) return false; const rad = parseFloat(getComputedStyle(e).borderTopLeftRadius); return rad / Math.min(r.width, r.height) >= 0.35; }).length;
  return { h1, legacy, panels, primaries: prim.length, unlit, overflow, capsules: caps };
})()`;
// live, signed in
const ctx = await b.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2, colorScheme: "dark", reducedMotion: "reduce" });
await ctx.addCookies([{ name: "vallo_first_run", value: "seen", url: BASE }]);
const p = await ctx.newPage(); p.setDefaultTimeout(30000);
await p.goto(`${BASE}/sign-in?next=%2Fprofile`, { waitUntil: "domcontentloaded" }); await p.waitForTimeout(3000);
await p.fill("#auth-email", process.env.QA_MEMBER_EMAIL);
await p.click("button:has-text('Continue')"); await p.waitForTimeout(8000);
await p.fill("#password", process.env.QA_MEMBER_PASSWORD).catch(() => {});
await p.click("button[type=submit]:has-text('Sign in')").catch(() => {});
await p.waitForTimeout(15000);
res.signedInLandedOn = new URL(p.url()).pathname;
for (const r of ["/host/rooms", "/host/photos", "/host/reservations", "/host/apply", "/styleguide"]) {
  for (const w of [390, 1440]) {
    await p.setViewportSize({ width: w, height: w === 390 ? 844 : 900 });
    let status = null;
    try { const resp = await p.goto(BASE + r, { waitUntil: "load", timeout: 45000 }); status = resp ? resp.status() : null; } catch (e) { status = "timeout"; }
    await p.waitForTimeout(2500);
    const m = await p.evaluate(MEASURE).catch(() => null);
    const name = r.replace(/\//g, "_").replace(/^_/, "");
    await p.screenshot({ path: `${dir}/live-${name}-${w}.jpg`, type: "jpeg", quality: 70, fullPage: true }).catch(() => {});
    res.live[`${r}@${w}`] = { status, finalPath: new URL(p.url()).pathname, ...m };
  }
}
await ctx.close();
// harness: later wizard steps
for (const step of ["business", "registration", "representative", "payout", "consent", "review"]) {
  for (const w of [390, 1440]) {
    const c = await b.newContext({ viewport: { width: w, height: w === 390 ? 844 : 900 }, deviceScaleFactor: w === 390 ? 2 : 1, colorScheme: "dark", reducedMotion: "reduce" });
    const q = await c.newPage();
    try { await q.goto(`${BASE}/preview/session-b/audit2-fixes?step=${step}`, { waitUntil: "load", timeout: 30000 }); } catch {}
    await q.waitForTimeout(2000);
    const m = await q.evaluate(MEASURE).catch(() => null);
    const shown = await q.evaluate(() => (document.body.innerText.match(/Step \d+ of \d+/) || [""])[0]).catch(() => "");
    await q.screenshot({ path: `${dir}/wizard-${step}-${w}.jpg`, type: "jpeg", quality: 70, fullPage: true }).catch(() => {});
    res.harness[`${step}@${w}`] = { progress: shown, ...m };
    await c.close();
  }
}
writeFileSync(out, JSON.stringify(res, null, 1));
await b.close();
console.log("done");
