// Independent audit of the Session B sweep: shoot + measure every harness route.
// node shoot.mjs <base> <routes.txt> <outdir>
import { chromium } from "/tmp/claude-0/-home-user-read-it-well/b3cd0ca7-526a-577f-99c1-fc3870877a0e/scratchpad/wt-audit/node_modules/playwright-core/index.mjs";
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";

const [BASE, LIST, OUT] = process.argv.slice(2);
mkdirSync(OUT, { recursive: true });
const routes = readFileSync(LIST, "utf8").split("\n").map((l) => l.trim()).filter((l) => l && !l.startsWith("#"));
const browser = await chromium.launch({
  executablePath: "/opt/pw-browsers/chromium",
  args: ["--enable-unsafe-swiftshader", "--use-angle=swiftshader"],
});

function slug(r) {
  return r.replace(/^\/preview\//, "").replace(/[/?=&]+/g, "_").replace(/_+$/, "") || "root";
}

const measure = () => {
  const out = { overflow: 0, small: [], capsules: [], primaries: [], flatSelected: [], legacy: [], oddContainers: [] };
  const de = document.documentElement;
  out.overflow = Math.max(de.scrollWidth, document.body.scrollWidth) - window.innerWidth;
  const vis = (el) => {
    const r = el.getBoundingClientRect();
    if (r.width < 1 || r.height < 1) return null;
    const cs = getComputedStyle(el);
    if (cs.visibility === "hidden" || cs.display === "none" || Number(cs.opacity) === 0) return null;
    return r;
  };
  const cls = (el) => (el.tagName.toLowerCase() + "." + (typeof el.className === "string" ? el.className : "").trim().split(/\s+/).slice(0, 4).join(".")).slice(0, 90);
  // small text
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  const seen = new Set();
  while (walker.nextNode()) {
    const n = walker.currentNode;
    const t = n.textContent.trim();
    if (!t || !n.parentElement) continue;
    const p = n.parentElement;
    if (seen.has(p)) continue;
    seen.add(p);
    if (p.closest("script,style,noscript,[aria-hidden='true'] svg")) continue;
    if (!vis(p)) continue;
    const fs = parseFloat(getComputedStyle(p).fontSize);
    if (fs < 11) out.small.push({ t: t.slice(0, 30), fs: +fs.toFixed(1), el: cls(p) });
  }
  // capsules on text-bearing controls
  const ctrlSel = "button,a,[role=button],[role=tab],[role=link],input:not([type=checkbox]):not([type=radio]):not([type=range]):not([type=hidden]):not([type=file]),select,textarea,summary,[class*=chip],[class*=badge],[class*=pill],[class*=tag],[class*=seg]";
  for (const el of document.querySelectorAll(ctrlSel)) {
    const r = vis(el);
    if (!r) continue;
    const isField = /^(INPUT|SELECT|TEXTAREA)$/.test(el.tagName);
    const txt = isField ? "field" : (el.innerText || "").trim();
    if (!txt) continue;
    const cs = getComputedStyle(el);
    const short = Math.min(r.width, r.height);
    const rad = (v) => (v.endsWith("%") ? (parseFloat(v) / 100) * short : parseFloat(v) || 0);
    const radius = Math.max(rad(cs.borderTopLeftRadius), rad(cs.borderTopRightRadius), rad(cs.borderBottomLeftRadius), rad(cs.borderBottomRightRadius));
    const drawn = cs.backgroundColor !== "rgba(0, 0, 0, 0)" || cs.backgroundImage !== "none" || parseFloat(cs.borderTopWidth) > 0 || cs.boxShadow !== "none";
    if (!drawn || radius === 0) continue;
    const ratio = Math.min(radius, short / 2) / short;
    if (ratio >= 0.35) out.capsules.push({ t: txt.slice(0, 24), ratio: +ratio.toFixed(2), w: Math.round(r.width), h: Math.round(r.height), r: +radius.toFixed(1), el: cls(el) });
  }
  // primaries
  for (const el of document.querySelectorAll(".nf-btn--primary")) {
    const r = vis(el);
    if (!r) continue;
    const cs = getComputedStyle(el);
    const lit = cs.backgroundImage.includes("gradient");
    const dis = el.disabled || el.getAttribute("aria-disabled") === "true";
    const grey = !lit || Number(cs.opacity) < 0.75 || /grayscale|saturate\(0/.test(cs.filter);
    out.primaries.push({ t: (el.innerText || el.getAttribute("aria-label") || "").trim().slice(0, 24), lit, opacity: cs.opacity, filter: cs.filter, dis, grey, el: cls(el) });
  }
  // unlit primaries not using nf-btn--primary but looking like primary: skip
  // selected states
  const selSel = "[aria-selected=true],[aria-pressed=true],[aria-current=page],[aria-current=true],[aria-current=step],[data-state=on],[data-state=active],[class*='--on'],[class*='--active'],[class*='is-active'],[role=radio][aria-checked=true],[role=tab][aria-selected=true]";
  for (const el of document.querySelectorAll(selSel)) {
    const r = vis(el);
    if (!r) continue;
    const cs = getComputedStyle(el);
    const bgc = cs.backgroundColor;
    const flat = cs.backgroundImage === "none" && bgc !== "rgba(0, 0, 0, 0)" && bgc !== "transparent";
    const m = bgc.match(/[\d.]+/g) || [];
    const [R, G, B, A = "1"] = m.map(Number);
    const navy = flat && B > R + 20 && B < 200 && Number(A) > 0.3;
    if (flat) out.flatSelected.push({ t: (el.innerText || "").trim().slice(0, 24), bg: bgc, navy, shadow: cs.boxShadow.slice(0, 60), el: cls(el) });
  }
  // legacy material classes still in markup
  for (const el of document.querySelectorAll(".nf-card:not(.nf-panel),.nf-glass--card,.nf-glass--tile,.nf-social-card,.nf-glyph-tile")) {
    if (!vis(el)) continue;
    out.legacy.push(cls(el));
  }
  // containers with their own radius (not the 10px container corner) drawing a box
  for (const el of document.querySelectorAll("body *")) {
    if (/^(IMG|VIDEO|SVG|PATH|CANVAS|PICTURE|INPUT|SELECT|TEXTAREA|BUTTON|A|LABEL)$/i.test(el.tagName)) continue;
    const r = el.getBoundingClientRect();
    if (r.width < 120 || r.height < 48) continue;
    const cs = getComputedStyle(el);
    if (cs.display === "none" || cs.visibility === "hidden") continue;
    const radius = parseFloat(cs.borderTopLeftRadius) || 0;
    if (radius < 12 || radius > 40) continue;
    const hasBox = (cs.backgroundColor !== "rgba(0, 0, 0, 0)" || cs.backgroundImage !== "none") && (parseFloat(cs.borderTopWidth) > 0 || cs.boxShadow !== "none");
    if (!hasBox) continue;
    const c = typeof el.className === "string" ? el.className : "";
    if (/nf-panel|nf-plate|nf-btn|nf-field|avatar|nf-sheet|nf-tabbar/.test(c)) continue;
    out.oddContainers.push({ r: radius, w: Math.round(r.width), h: Math.round(r.height), el: cls(el) });
  }
  out.smallCount = out.small.length;
  out.small = out.small.slice(0, 12);
  out.oddContainers = out.oddContainers.slice(0, 15);
  out.legacyCount = out.legacy.length;
  out.legacy = [...new Set(out.legacy)].slice(0, 10);
  return out;
};

const results = {};
const widths = (process.env.WIDTHS || "390,1440").split(",").map(Number);
for (const route of routes) {
  results[route] = {};
  for (const w of widths) {
    const ctx = await browser.newContext({
      viewport: { width: w, height: w === 390 ? 844 : 900 },
      deviceScaleFactor: w === 390 ? 2 : 1,
      colorScheme: "dark",
      reducedMotion: "reduce",
    });
    const page = await ctx.newPage();
    let status = 0;
    try {
      const resp = await page.goto(BASE + route, { waitUntil: "load", timeout: 30000 });
      status = resp?.status() ?? 0;
    } catch (e) {
      status = -1;
    }
    await page.waitForTimeout(1500);
    // scroll to fire reveals
    await page.evaluate(async () => {
      for (let y = 0; y < document.body.scrollHeight; y += 400) { window.scrollTo(0, y); await new Promise((r) => setTimeout(r, 40)); }
      window.scrollTo(0, 0);
    });
    await page.waitForTimeout(300);
    let m = null;
    try { m = await page.evaluate(measure); } catch (e) { m = { error: String(e) }; }
    m.status = status;
    m.finalUrl = page.url().replace(BASE, "");
    m.theme = await page.evaluate(() => document.documentElement.getAttribute("data-theme"));
    const file = `${OUT}/${slug(route)}-${w}.jpg`;
    try { await page.screenshot({ path: file, fullPage: true, type: "jpeg", quality: 70 }); } catch (e) { m.shotError = String(e).slice(0, 120); }
    results[route][w] = m;
    writeFileSync(`${OUT}/m-${slug(route)}-${w}.json`, JSON.stringify(m));
    await ctx.close();
    process.stdout.write(`${status} ${w} ${route} ovf=${m.overflow} small=${m.smallCount} caps=${m.capsules?.length} flatSel=${m.flatSelected?.length} legacy=${m.legacyCount} odd=${m.oddContainers?.length} grey=${m.primaries?.filter((p) => p.grey).length}\n`);
  }
}
writeFileSync(`${OUT}/measure.json`, JSON.stringify(results, null, 1));
await browser.close();
